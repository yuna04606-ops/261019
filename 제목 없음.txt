<script>
  const $ = id => document.getElementById(id);
  let selectedFile = null;
  let currentQuestions = [];

  // API 키는 이 브라우저에만 저장
  function loadKey() {
    const key = localStorage.getItem('studynote_gemini_key') || '';
    $('apiKey').value = key;
    updateStatus(!!key);
  }

  function updateStatus(saved) {
    $('apiStatus').textContent = saved ? '● API 설정 완료' : '● API 설정 필요';
    $('apiStatus').style.color = saved ? '#218653' : '#bd7b21';
  }

  $('saveKey').addEventListener('click', () => {
    const key = $('apiKey').value.trim();
    if (!key) return showError('먼저 Gemini API 키를 입력해 주세요.');
    localStorage.setItem('studynote_gemini_key', key);
    updateStatus(true);
    hideError();
    alert('API 키를 이 브라우저에 저장했습니다.');
  });

  $('deleteKey').addEventListener('click', () => {
    localStorage.removeItem('studynote_gemini_key');
    $('apiKey').value = '';
    updateStatus(false);
    alert('브라우저에 저장된 API 키를 삭제했습니다.');
  });

  $('apiKey').addEventListener('input', () => {
    updateStatus(!!localStorage.getItem('studynote_gemini_key'));
  });

  $('pdfFile').addEventListener('change', e => setFile(e.target.files[0]));

  function setFile(file) {
    if (!file) return;
    hideError();

    if (file.type !== 'application/pdf' && !file.name.toLowerCase().endsWith('.pdf')) {
      return showError('PDF 파일만 업로드할 수 있습니다.');
    }
    if (file.size > 15 * 1024 * 1024) {
      return showError('파일이 너무 큽니다. 15MB 이하의 PDF를 선택해 주세요.');
    }

    selectedFile = file;
    $('fileInfo').style.display = 'block';
    $('fileInfo').textContent =
      '📎 ' + file.name + ' · ' + (file.size / 1024 / 1024).toFixed(2) + 'MB';
    $('results').style.display = 'none';
  }

  const dropzone = $('dropzone');
  dropzone.addEventListener('dragover', e => {
    e.preventDefault();
    dropzone.style.borderColor = '#5b5bd6';
  });
  dropzone.addEventListener('dragleave', () => {
    dropzone.style.borderColor = '';
  });
  dropzone.addEventListener('drop', e => {
    e.preventDefault();
    dropzone.style.borderColor = '';
    const file = e.dataTransfer.files[0];
    setFile(file);
  });

  $('analyzeBtn').addEventListener('click', () => analyze(false));
  $('retryBtn').addEventListener('click', () => analyze(true));

  function showError(message) {
    $('error').textContent = message;
    $('error').style.display = 'block';
  }

  function hideError() {
    $('error').style.display = 'none';
    $('error').textContent = '';
  }

  function setLoading(loading) {
    $('loading').style.display = loading ? 'block' : 'none';
    $('analyzeBtn').disabled = loading;
    $('retryBtn').disabled = loading;
  }

  function readFileAsBase64(file) {
    return new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onload = () => {
        const result = String(reader.result);
        resolve(result.substring(result.indexOf(',') + 1));
      };
      reader.onerror = () => reject(new Error('PDF 파일을 읽지 못했습니다.'));
      reader.readAsDataURL(file);
    });
  }

  async function analyze(retry) {
    hideError();

    const apiKey = (
      localStorage.getItem('studynote_gemini_key') ||
      $('apiKey').value
    ).trim();

    if (!apiKey) return showError('Gemini API 키를 먼저 입력하고 저장해 주세요.');
    if (!selectedFile) return showError('분석할 PDF 파일을 먼저 선택해 주세요.');

    if (!$('apiKey').value.trim()) $('apiKey').value = apiKey;

    setLoading(true);
    $('results').style.display = 'none';

    try {
      const base64 = await readFileAsBase64(selectedFile);
      const prompt = `
너는 대학 강의 교안을 분석하는 학습 전문가다.
첨부된 PDF의 실제 내용만 근거로 한국어 학습자료를 작성해라.
교안에 없는 사실이나 개념을 만들어내지 마라.
텍스트가 불명확하거나 근거가 부족한 부분은 추측하지 마라.

다음 JSON 형식으로만 응답해라.
{
  "summary": [
    {"heading": "주제", "content": "핵심 내용을 구체적으로 설명"}
  ],
  "questions": [
    {
      "question": "문제 내용",
      "options": ["선택지1", "선택지2", "선택지3", "선택지4"],
      "answer": 1,
      "explanation": "교안 내용을 근거로 정답인 이유를 설명"
    }
  ]
}

작성 규칙:
1. summary는 중요한 개념, 정의, 특징, 절차 등을 5~8개 항목으로 구성한다.
2. 각 항목은 시험 공부에 도움이 되도록 구체적으로 정리한다.
3. questions는 정확히 10개 작성한다.
4. 모든 문제는 객관식 4지선다이며 options 배열에는 선택지 4개만 넣는다.
5. answer는 정답 선택지의 배열 인덱스다. 0부터 3까지의 정수로 작성한다.
6. 문제는 교안의 주요 개념을 골고루 다루고, 선택지끼리 중복되지 않게 작성한다.
7. 정답이 하나로 명확한 문제를 만들고, 오답은 그럴듯하지만 교안 내용상 틀리게 구성한다.
8. explanation은 정답 근거를 설명하고, 가능하면 관련 개념도 간단히 설명한다.
9. PDF에서 확인할 수 없는 내용을 정답의 근거로 삼지 마라.
10. 교안 내용을 충분히 확인할 수 없다면 summary에 그 한계를 명시하고, 근거 없는 문제를 억지로 만들지 마라.
${retry ? '이전과 가능한 한 겹치지 않도록 다른 개념과 문제 유형을 선택해라.' : ''}
`;

      const response = await fetch(
        'https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key='
          + encodeURIComponent(apiKey),
        {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            contents: [{
              parts: [
                { text: prompt },
                { inlineData: { mimeType: 'application/pdf', data: base64 } }
              ]
            }],
            generationConfig: {
              responseMimeType: 'application/json',
              temperature: 0.4
            }
          })
        }
      );

      const data = await response.json();

      if (!response.ok) {
        const apiMessage = data?.error?.message || 'API 요청에 실패했습니다.';
        if (response.status === 400 || response.status === 403) {
          throw new Error('Gemini API 요청 오류: ' + apiMessage +
            '\nAPI 키가 유효한지, API 사용 설정 및 할당량을 확인해 주세요.');
        }
        if (response.status === 429) {
          throw new Error('API 사용 한도에 도달했습니다. 잠시 후 다시 시도해 주세요.');
        }
        throw new Error(apiMessage);
      }

      const parts = data?.candidates?.[0]?.content?.parts || [];
      const raw = parts.map(p => p.text || '').join('').trim();
      if (!raw) {
        throw new Error('AI가 결과를 반환하지 않았습니다. PDF 파일이나 API 응답을 확인해 주세요.');
      }

      const cleaned = raw.replace(/^```(?:json)?\s*/i, '').replace(/\s*```$/, '');
      const result = JSON.parse(cleaned);

      if (!Array.isArray(result.summary) || !Array.isArray(result.questions)) {
        throw new Error('응답 형식이 올바르지 않습니다. 다시 생성해 주세요.');
      }
      if (result.questions.length !== 10) {
        throw new Error('예상문제가 10개 생성되지 않았습니다. 다시 시도해 주세요.');
      }

      result.questions.forEach((q, i) => {
        if (
          !q.question ||
          !Array.isArray(q.options) ||
          q.options.length !== 4 ||
          !Number.isInteger(q.answer) ||
          q.answer < 0 ||
          q.answer > 3
        ) {
          throw new Error((i + 1) + '번 문제의 형식이 올바르지 않습니다. 다시 생성해 주세요.');
        }
      });

      renderResults(result);
    } catch (err) {
      showError('분석 중 문제가 발생했습니다.\n' +
        (err.message || '알 수 없는 오류') +
        '\n\n문제가 계속되면 API 키, 인터넷 연결, PDF 파일 크기를 확인해 주세요.');
    } finally {
      setLoading(false);
    }
  }

  function renderResults(result) {
    $('resultFilename').textContent = selectedFile.name;
    const summary = $('summary');
    summary.replaceChildren();

    result.summary.forEach(item => {
      const section = document.createElement('section');
      const heading = document.createElement('h3');
      const content = document.createElement('p');
      heading.textContent = item.heading || '핵심 내용';
      content.textContent = item.content || '';
      section.append(heading, content);
      summary.appendChild(section);
    });

    currentQuestions = result.questions;
    const container = $('questions');
    container.replaceChildren();

    currentQuestions.forEach((q, i) => {
      const article = document.createElement('article');
      article.className = 'question';

      const title = document.createElement('div');
      title.className = 'q-title';
      title.textContent = `Q${String(i + 1).padStart(2, '0')}. ${q.question}`;
      article.appendChild(title);

      q.options.forEach((option, j) => {
        const label = document.createElement('label');
        label.className = 'option';

        const radio = document.createElement('input');
        radio.type = 'radio';
        radio.name = 'question-' + i;
        radio.value = String(j);

        const text = document.createElement('span');
        text.textContent = `${j + 1}. ${option}`;

        label.append(radio, text);
        article.appendChild(label);
      });

      const explanation = document.createElement('div');
      explanation.className = 'explanation';
      explanation.id = 'explanation-' + i;
      explanation.textContent =
        `정답: ${q.answer + 1}번. ${q.options[q.answer]}\n해설: ${q.explanation || '별도 해설이 없습니다.'}`;
      article.appendChild(explanation);
      container.appendChild(article);
    });

    $('score').textContent = '';
    $('gradeBtn').disabled = false;
    $('gradeBtn').style.display = 'block';
    $('results').style.display = 'block';
    $('results').scrollIntoView({ behavior: 'smooth', block: 'start' });
  }

  $('gradeBtn').addEventListener('click', () => {
    let score = 0;
    let unanswered = 0;

    currentQuestions.forEach((q, i) => {
      const selected = document.querySelector(`input[name="question-${i}"]:checked`);
      const options = document.querySelectorAll(`input[name="question-${i}"]`);

      if (!selected) unanswered++;
      if (selected && Number(selected.value) === q.answer) score++;

      options.forEach((radio, j) => {
        radio.disabled = true;
        const label = radio.closest('.option');
        if (j === q.answer) label.classList.add('correct');
        if (radio.checked && j !== q.answer) label.classList.add('wrong');
      });

      $('explanation-' + i).classList.add('show');
    });

    $('score').textContent =
      `점수: ${score * 10}점 (${score}/10 정답)` +
      (unanswered ? ` · 미응답 ${unanswered}문제` : '');
    $('gradeBtn').disabled = true;
    $('gradeBtn').textContent = '채점 완료';
  });

  loadKey();
</script>
